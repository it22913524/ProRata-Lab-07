public class IT22913524Lab7Q2 {
    public static void main(String[] args) {

        // ---------- Part (a): $ figure (4 rows, 5 columns) ----------
        for (int row = 1; row <= 4; row++) {
            for (int col = 1; col <= 5; col++) {
                System.out.print(col < 5 ? "$ " : "$");
            }
            System.out.println();
        }

        System.out.println();

        // ---------- Part (b): numbers 1-5 with stars ----------
        for (int i = 1; i <= 5; i++) {
            System.out.print(i + " -");
            for (int j = 1; j <= i; j++) {
                System.out.print(" *");
            }
            System.out.println();
        }

        System.out.println();

        // ---------- Part (c): 55555 / 4444 / 333 / 22 / 1 ----------
        for (int i = 5; i >= 1; i--) {
            for (int j = 1; j <= i; j++) {
                System.out.print(i);
            }
            System.out.println();
        }
    }
}
